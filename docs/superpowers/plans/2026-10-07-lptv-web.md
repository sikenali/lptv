# LX TV Web 版实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 pytv（Python + pywebview 桌面客户端）复刻为 Web 版本，保留 inject.js 全部前端逻辑，通过寄生式取流 + HLS 同源化代理实现录制/截图功能。

**Architecture:** Node.js Express 服务端 + Playwright 有头浏览器实例（寄生式 hook Hls.prototype.loadSource 捕获 m3u8），hlsProxy.ts 改写 m3u8 分片 URL 为本服务实现同源化（解除 canvas 污染），epgProxy.ts 代理 EPG 请求避免 CORS。inject.js 通过 adapter.js 透明拦截 pywebview API → fetch。Cookie 文件持久化，开机自启替代为 lastPid 续播。

**Tech Stack:** TypeScript, Express, Playwright, hls.js, MediaRecorder API

## Global Constraints

- 前端 inject.js **零修改**，通过 adapter.js 提供兼容层
- 端口默认 3456，可通过 PORT 环境变量覆盖
- Cookie 存储路径：`data/cookies.json`（Playwright 标准格式）
- EPG 缓存 TTL：30 分钟
- 频道数据完整移植自 pytv 的 channels.py（CCTV 1-17 + CGTN + 30+ 卫视），CCTV-1 与 CCTV-14 的 cnlid 重复（2024078201），以 pid 为主键
- 录制/截图前提：HLS 流必须同源化（hlsProxy.ts 改写分片 URL 走 /api/seg/?u=）
- 换台方式：Playwright 点击 `.tv-main-con-r-list-left > div[data-pid="xxx"]`，4s 兜底
- 开机自启替代方案：记住 lastPid，打开页面即续播
- 黑帧看门狗：requestVideoFrameCallback + 16×16 canvas 采样

```typescript
  res?: Reservation[];

function readState(): UiState {
  if (!existsSync(STATE_FILE)) return {};
  try {
    return JSON.parse(readFileSync(STATE_FILE, 'utf-8'));
  } catch {
    return {};
  }
}

function writeState(state: UiState) {
  if (!existsSync(DATA_DIR)) {
    mkdirSync(DATA_DIR, { recursive: true });
  }
  writeFileSync(STATE_FILE, JSON.stringify(state, null, 2), 'utf-8');
}

export const StateManager = {
  async load(): Promise<UiState> {
    return readState();
  },

  async save(patch: Partial<UiState>): Promise<void> {
    const state = readState();
    Object.assign(state, patch);
    writeState(state);
  }
};
```

```typescript
import { chromium, Browser, BrowserContext, Page } from 'playwright';
import { existsSync, readFileSync } from 'fs';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const COOKIES_FILE = join(__dirname, '../../data/cookies.json');
const OFFICIAL_BASE = 'https://www.yangshipin.cn';
const API_BASE = 'https://capi.yangshipin.cn';

export interface EpgProgram {
  id: string;
  name: string;
  start: string;
  end: string;
  s0: number;
  e0: number;
  dur: number;
}

// Hook 注入代码：拦截 Hls.prototype.loadSource 捕获 m3u8 URL
const HOOK_SCRIPT = `
(function() {
  if (window.__lxHookInstalled) return;
  window.__lxHookInstalled = true;
  window.__lxPlayUrl = '';
  window.__lxLastPid = '';

  var OrigHlsLoad = window.Hls && window.Hls.prototype && window.Hls.prototype.loadSource;
  if (OrigHlsLoad) {
    window.Hls.prototype.loadSource = function(url) {
      window.__lxPlayUrl = url;
      window.__lxLastPid = (new URL(url, location.href).searchParams.get('pid')) || '';
      return OrigHlsLoad.call(this, url);
    };
  }

  // 也 hook XHR/fetch 拦截 get_live_info
  var origOpen = XMLHttpRequest.prototype.open;
  XMLHttpRequest.prototype.open = function(method, url) {
    if (String(url).indexOf('get_live_info') >= 0) {
      this.addEventListener('load', function() {
        try {
          var d = JSON.parse(this.responseText).data;
          if (d && d.iretcode === 0 && d.playurl) {
            window.__lxPlayUrl = d.playurl;
          }
        } catch(e) {}
      });
    }
    return origOpen.apply(this, arguments);
  };
})();
`;

class YangshipinBrowser {
  private browser: Browser | null = null;
  private context: BrowserContext | null = null;
  private _page: Page | null = null;
  private cookiesLoaded = false;
  private currentPid = '';

  get page(): Page {
    if (!this._page) throw new Error('Browser not initialized. Call init() first.');
    return this._page;
  }

  async init() {
    if (this.browser) return true;
    if (!existsSync(COOKIES_FILE)) return false;

    try {
      const state = JSON.parse(readFileSync(COOKIES_FILE, 'utf-8'));
      this.browser = await chromium.launch({ headless: false });
      this.context = await this.browser.newContext({
        ...state,
        extraHTTPHeaders: {
          'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/132.0.0.0 Safari/537.36',
        },
      });
      this._page = await this.context.newPage();
      this.cookiesLoaded = true;
      return true;
    } catch (e) {
      console.error('[Playwright] Failed to init browser:', e);
      this.cookiesLoaded = false;
      return false;
    }
  }

  isLoggedIn(): boolean {
    return this.cookiesLoaded && this._page !== null;
  }

  // 寄生式获取流地址：加载页面 → 注入 hook → 等待 m3u8 URL
  async getPlayUrl(pid: string): Promise<string | null> {
    if (!this._page) await this.init();
    if (!this._page) return null;

    this.currentPid = pid;
    const url = `${OFFICIAL_BASE}/tv/home?pid=${pid}`;

    try {
      // 导航到频道页面
      await this._page.goto(url, { waitUntil: 'domcontentloaded', timeout: 30000 });

      // 注入 hook 脚本
      await this._page.addInitScript(HOOK_SCRIPT);

      // 等待 Hls 实例加载 m3u8（最多 15 秒）
      const playUrl = await this._page.waitForFunction(
        () => (window as any).__lxPlayUrl || '',
        { timeout: 15000 }
      ).catch(() => null);

      if (playUrl) {
        console.log(`[Playwright] Captured playUrl for pid=${pid}: ${String(playUrl).slice(0, 80)}...`);
        return String(playUrl);
      }

      // 备选：直接调 get_live_info API
      return await this.getPlayUrlViaApi(pid);
    } catch (e) {
      console.error(`[Playwright] getPlayUrl(${pid}) failed:`, e);
      return null;
    }
  }

  // 备选：通过 API 直接取流
  private async getPlayUrlViaApi(pid: string): Promise<string | null> {
    if (!this._page) return null;

        const json = await r.json() as { data: { iretcode: number; playurl: string } };

      }, pid);

    } catch (e) {

      return null;
    }
  }

  // 切换频道：点击官方侧栏按钮 + 4s 兜底
  async switchChannel(pid: string): Promise<boolean> {
    if (!this._page) return false;
    try {
      // 等待侧栏加载
      await this._page.waitForSelector('.tv-main-con-r-list-left', { timeout: 10000 }).catch(() => null);

      // 点击对应按钮
      const clicked = await this._page.evaluate((p: string) => {
        const btn = document.querySelector(`.tv-main-con-r-list-left > div[data-pid="${p}"]`);
        if (btn) {
          btn.click();
          return true;
        }
        return false;
      }, pid);

      if (!clicked) {
        console.warn(`[Playwright] switchChannel(${pid}): button not found`);
        return false;
      }

      // 等待新流地址（最多 4 秒）
      this._page.evaluate(() => { (window as any).__lxPlayUrl = ''; });
      const playUrl = await this._page.waitForFunction(
        () => (window as any).__lxPlayUrl || '',
        { timeout: 4000 }
      ).catch(() => null);

      if (!playUrl) {
        // 4s 兜底：用自愈后的真实 pid 再点一次
        console.warn(`[Playwright] switchChannel(${pid}): no playUrl in 4s, retrying`);
        const secondPid = await this._page.evaluate(() => (window as any).__lxLastPid || '');
        if (secondPid && secondPid !== pid) {
          await this._page.evaluate((p: string) => {
            const btn = document.querySelector(`.tv-main-con-r-list-left > div[data-pid="${p}"]`);
            if (btn) btn.click();
          }, secondPid);
        }
      }

      this.currentPid = pid;
      return true;
    } catch (e) {
      console.error(`[Playwright] switchChannel(${pid}) failed:`, e);
      return false;
    }

  async getEpg(pid: string, ymd: string): Promise<EpgProgram[]> {

    }

    this.context = null;
    this._page = null;
    this.cookiesLoaded = false;
  }

// Protobuf parser（从 inject.js 移植）
function parseEpg(buf: Buffer): EpgProgram[] {
  const programs: EpgProgram[] = [];
  try {
    const bytes = new Uint8Array(buf);
    let i = 0;
    function varint() {
      let v = 0, s = 0, k = 0;
      while (i < bytes.length && k < 5) {
        const b = bytes[i++];
        v += (b & 0x7f) * Math.pow(2, s);
        s += 7;
        k++;
        if (!(b & 0x80)) break;
      }
      return v;
    }
    while (i < bytes.length) {
      const tag = varint();
      if (tag === 0) break;
      const f = tag >>> 3, wt = tag & 7;
      if (wt === 2) {
        const len = varint();
        const p = parseEpgEntry(bytes, i, i + len);
        if (p) programs.push(p);
        i += len;
      } else if (wt === 0) {
        varint();
      } else {
        i += varint();
      }
      if (programs.length > 200) break;
    }
  } catch (e) {}
  return programs;
}

function parseEpgEntry(bytes: Uint8Array, a: number, b: number): EpgProgram | null {
  let i = a, p: Partial<EpgProgram> = {};
  function varint() {
    let v = 0, s = 0, k = 0;
    while (i < b && k < 5) {
      const x = bytes[i++];
      v += (x & 0x7f) * Math.pow(2, s);
      s += 7;
      k++;
      if (!(x & 0x80)) break;
    }
    return v;
  }
  try {
    while (i < b) {
      const tag = varint();
      if (tag === 0) break;
      const f = tag >>> 3, wt = tag & 7;
      if (wt === 2) {
        const len = varint();
        if (i + len > b) break;
        const str = bytes.subarray(i, i + len).toString('utf-8');
        if (f === 1) p.id = str;
        else if (f === 2) p.name = str;
        else if (f === 5) p.start = str;
        else if (f === 6) p.end = str;
        i += len;
      } else if (wt === 0) {
        const v = varint();
        if (f === 3) p.s0 = v;
        else if (f === 4) p.e0 = v;
        else if (f === 7) p.dur = v;
      } else {
        i += varint();
      }
    }
  } catch (e) {}
  return p.name ? p as EpgProgram : null;
}
```

```typescript
import { Router, Request, Response } from 'express';
import { ygbrowser } from './playwright.js';

export const hlsRouter = Router();

// GET /api/manifest?pid=xxx
hlsRouter.get('/manifest', async (req: Request, res: Response) => {
  const pid = req.query.pid as string;
  if (!pid) {
    res.status(400).json({ error: 'pid required' });
    return;
  }

  try {
    // 从浏览器页面获取原始 m3u8（携带官方站 Cookie）
    const playUrl = await ygbrowser.getPlayUrl(pid);
    if (!playUrl) {
      res.status(502).json({ error: 'Failed to get play URL. Try refreshing cookies.' });
      return;
    }

    const m3u8Content = await ygbrowser.page.evaluate(async (url: string) => {
      const resp = await fetch(url);
      return await resp.text();
    }, playUrl);

    if (!m3u8Content) {
      res.status(502).json({ error: 'Failed to fetch manifest' });
      return;
    }

    // 改写 m3u8：将所有分片 URL 指向本服务
    const rewritten = rewriteManifest(m3u8Content, playUrl);

    res.set('Content-Type', 'application/vnd.apple.mpegurl');
    res.set('Access-Control-Allow-Origin', '*');
    res.send(rewritten);
  } catch (e) {
    console.error('[HLS] Manifest rewrite failed:', e);
    res.status(500).json({ error: 'Manifest processing failed' });
  }
});

// GET /api/seg/?u=<base64_url>&type=ts|key
hlsRouter.get('/seg', async (req: Request, res: Response) => {
  const u = req.query.u as string;
  const type = req.query.type as 'ts' | 'key' | undefined;

  if (!u) {
    res.status(400).json({ error: 'u parameter required' });
    return;
  }

  try {
    // base64 解码原始 URL
    const originalUrl = Buffer.from(u, 'base64').toString('utf-8');

    // 从浏览器页面代理请求（携带官方站 Cookie）
    const buffer = await ygbrowser.page.evaluate(async (url: string) => {
      const resp = await fetch(url);
      const buf = await resp.arrayBuffer();
      return Array.from(new Uint8Array(buf));
    }, originalUrl);

    const data = Buffer.from(buffer);

    // 如果是密钥，返回 AES-128 密钥格式
    if (type === 'key') {
      res.set('Content-Type', 'application/octet-stream');
      res.set('Access-Control-Allow-Origin', '*');
      res.send(data);
      return;
    }

    // TS 分片
    res.set('Content-Type', 'video/mp2t');
    res.set('Access-Control-Allow-Origin', '*');
    res.send(data);
  } catch (e) {
    console.error('[HLS] Segment proxy failed:', e);
    res.status(502).json({ error: 'Segment fetch failed' });
  }
});

/**
 * 改写 m3u8：将所有分片 URL 替换为本服务的 /api/seg/?u= 代理
 * 处理相对路径、绝对路径、EXT-X-KEY
 */
function rewriteManifest(m3u8: string, originalUrl: string): string {
  const { URL } = require('url');
  const baseUrl = new URL(originalUrl);
  const baseOrigin = `${baseUrl.protocol}//${baseUrl.hostname}${baseUrl.port ? ':' + baseUrl.port : ''}`;

  return m3u8
    // 改写 EXT-X-KEY 的 URI
    .replace(/#EXT-X-KEY:(.*)URI="([^"]+)"/g, (_match: string, prefix: string, keyUrl: string) => {
      const encoded = Buffer.from(encodeAbsoluteUrl(keyUrl, baseOrigin)).toString('base64');
      return `#EXT-X-KEY:${prefix}URI="/api/seg?u=${encoded}&type=key"`;
    })
    // 改写分片 URL（#EXTINF 后的行）
    .replace(/(#EXTINF[^\\r\\n]*[\\r\\n]+)([^#])/g, (_match: string, header: string, urlLine: string) => {
      const url = urlLine.trim();
      if (!url || url.startsWith('#')) return _match;
      const absoluteUrl = encodeAbsoluteUrl(url, baseOrigin);
      const encoded = Buffer.from(absoluteUrl).toString('base64');
      return `${header}/api/seg?u=${encoded}`;
    });
}

function encodeAbsoluteUrl(url: string, baseOrigin: string): string {
  if (url.startsWith('http')) return url;
  if (url.startsWith('/')) return baseOrigin + url;
  return baseOrigin + '/' + url;
}
```

```typescript
import { Router, Request, Response } from 'express';
import { ygbrowser, EpgProgram } from './playwright.js';

const CACHE_TTL_MS = 30 * 60 * 1000; // 30 分钟
const EPG_DAY_MAX = 5;

interface CacheEntry {
  data: EpgProgram[];
  expiry: number;
}

export const epgRouter = Router();

// GET /api/epg?pid=xxx&ymd=xxxxxx
epgRouter.get('/', async (req: Request, res: Response) => {
  const pid = req.query.pid as string;
  const ymd = req.query.ymd as string;
  if (!pid || !ymd) {
    res.status(400).json({ error: 'pid and ymd required' });
    return;
  }

  const programs = await fetchEpg(pid, ymd);
  res.json(programs);
});

// GET /api/epg/probe?pid=xxx
epgRouter.get('/probe', async (req: Request, res: Response) => {
  const pid = req.query.pid as string;
  if (!pid) {
    res.status(400).json({ error: 'pid required' });
    return;
  }
  const days = await probeAvailableDays(pid);
  res.json(days);
});

// 内存缓存
const cache = new Map<string, CacheEntry>();

async function fetchEpg(pid: string, ymd: string): Promise<EpgProgram[]> {
  const key = `${pid}|${ymd}`;
  const cached = cache.get(key);
  if (cached && cached.expiry > Date.now()) {
    return cached.data;
  }

  const data = await ygbrowser.getEpg(pid, ymd);
  if (data.length > 0) {
    cache.set(key, {
      data,
      expiry: Date.now() + CACHE_TTL_MS,
    });
  }
  return data;
}

async function probeAvailableDays(pid: string): Promise<number[]> {
  if (!/^\d+$/.test(pid)) return [0];
  const available: number[] = [];
  const today = new Date();

  const todayYmd = toYmd(today);
  const todayData = await fetchEpg(pid, todayYmd);
  if (todayData.length > 0) available.push(0);

  for (const dir of [1, -1] as const) {
    for (let off = dir; Math.abs(off) <= EPG_DAY_MAX; off += dir * 2) {
      const d = new Date(today);
      d.setDate(d.getDate() + off);
      const ymd = toYmd(d);
      const data = await fetchEpg(pid, ymd);
      if (data.length > 0) {
        available.push(off);
      }
    }
  }
  return available.sort((a, b) => a - b);
}

function toYmd(date: Date): string {
  return date.getFullYear() +
    String(date.getMonth() + 1).padStart(2, '0') +
    String(date.getDate()).padStart(2, '0');
}
```

### Task 9: Cookie 初始化脚本

**Files:**
- Create: `scripts/init-cookies.js`

**Interfaces:**

- [ ] **Step 1: 创建 scripts/init-cookies.js**

```javascript
/* Cookie 初始化脚本
 * 启动有头浏览器，引导用户登录 yangshipin.cn
 * 登录完成后自动保存 Cookie 到 data/cookies.json
 */
const { chromium } = require('playwright');
const fs = require('fs');
const path = require('path');

async function main() {
  const cookiesFile = path.join(__dirname, '../data/cookies.json');
  console.log('[Init] Starting browser for yangshipin.cn login...');
  console.log('[Init] Please login manually, then close this browser window.');
  console.log('[Init] Cookie file:', cookiesFile);

  const browser = await chromium.launch({ headless: false });
  const context = await browser.newContext();
  const page = await context.newPage();

  await page.goto('https://www.yangshipin.cn/tv/home');
  console.log('[Init] Opened yangshipin.cn, please login...');

  // 等待用户手动登录（用户关闭页面或超时）
  await page.waitForEvent('close', { timeout: 0 }).catch(() => {});

  // 保存 Cookie
  const state = await context.storageState();
  fs.mkdirSync(path.join(__dirname, '../data'), { recursive: true });
  fs.writeFileSync(cookiesFile, JSON.stringify(state, null, 2));
  console.log('[Init] Cookies saved to', cookiesFile);

  await browser.close();
}

main().catch(err => {
  console.error('[Init] Error:', err.message);
  process.exit(1);
});
```
